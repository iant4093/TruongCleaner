## New (`Sources/CleanerCore/Policy/` and `Sources/CleanerCore/Match/`)

```swift
/// Values the pure layer needs from the outside world.
public struct Environment: Sendable {
    public var home: URL                 // tests pass a fixture home
    public var darwinUserCache: URL?
    public var darwinUserTemp: URL?
    public var now: Date
}

public enum RootExpansion {
    /// Expands a RootSpec to concrete folder URLs. "*" segments are returned as patterns
    /// for the scanner to resolve (`ExpandedRoot.pattern`), never resolved here.
    public static func expand(_ spec: RootSpec, in env: Environment) -> ExpandedRoot?   // nil if the base is unknown
}

public enum ProtectedReason: Sendable, Hashable { case system, root, appleData, syncAndIdentity, fragileAppleCache,
    agentState, liveStore, licensedVendor, passwordManager, userExcluded }

public struct PathPolicy: Sendable {
    public init(environment: Environment, userExclusions: [URL])
    /// Paths must already be standardized and have their parent's real path resolved by the caller.
    public func protectedReason(for url: URL) -> ProtectedReason?
    public func isInside(_ url: URL, roots: [URL]) -> Bool          // component-wise, never string prefix
    public func isRoot(_ url: URL, roots: [URL]) -> Bool            // equal to a root, ~, ~/Library, /, /Applications, /Library, /Users, /Users/Shared
}

/// One candidate as the scanner sees it (depth relative to the expanded root).
public struct Candidate: Sendable { url, depth, id: (fsid, fileID), isDirectory, isSymlink, flags, allocatedBytes, privateBytes, itemCount, newestModification: Date? }

public enum MatchOutcome: Sendable { case finding(Finding), skipped(Candidate, SkipReason) }
public enum SkipReason: String, Sendable, Codable { case outsideRoots, protectedPath, isRoot, tooNew, keepNewest,
    unknownAge, excludedName, symlinkNotAllowed, missingSibling, infoRule }

public enum Matcher {
    /// Applies a rule's match / depth / guards (minAge, keepNewest, excludeNames, requiresSibling)
    /// and the PathPolicy to candidates found under one expanded root.
    public static func evaluate(rule: Rule, root: URL, candidates: [Candidate],
                                siblingNames: [URL: Set<String>],   // per candidate parent: names present
                                policy: PathPolicy, env: Environment) -> [MatchOutcome]
}
```

Rules:

- The global protected list is `docs/RULE_CATALOG.md` §3, encoded as data in
  `Policy/ProtectedList.swift` (pattern + reason), with `~` relative to
  `env.home`. Patterns support a trailing `/**` (subtree), `*` within one
  component, and exact paths. The `~/.claude/**`-with-exceptions rows become
  "protected except these child names".
- `isInside` compares path components after `standardizedFileURL`; `"/a/bc"` is
  not inside `"/a/b"`.
- `minAge`: an item with `newestModification == nil` is skipped with
  `unknownAge` (unknown means keep). Age is `env.now - newestModification`.
- `keepNewest N`: group by parent folder; keep the N with the newest
  `newestModification` (nil sorts as newest, so it is kept).
- Info-tier rules always return `.skipped(.infoRule)` for every candidate.
- Symlinks are skipped (`symlinkNotAllowed`) unless the match is `.children`
  and the candidate is the symlink itself (delete-as-link is still decided later
  by the gate; here allow it only for Safe/Review with `.children`).
- A candidate equal to the root is `.isRoot`. Depth must be ≥ 1.
- Order of checks: info rule → isRoot → outsideRoots → protectedPath → symlink
  → excludedName → requiresSibling → age → keepNewest (last, across survivors).
